cmake_minimum_required(VERSION 3.18)
project(npuforge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NPUFORGE_PYTHON "Build Python bindings" ON)
add_library(npuforge_sim STATIC cpp/src/simulator.cpp)
set_target_properties(npuforge_sim PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(npuforge_sim PUBLIC cpp/include)
target_compile_options(npuforge_sim PRIVATE -Wall -Wextra -Wpedantic)
if(NPUFORGE_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_native cpp/src/bindings.cpp)
  target_link_libraries(_native PRIVATE npuforge_sim)
  install(TARGETS _native LIBRARY DESTINATION npuforge)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(simulator_tests cpp/tests/test_simulator.cpp)
  target_link_libraries(simulator_tests PRIVATE npuforge_sim)
  add_test(NAME simulator_invariants COMMAND simulator_tests)
endif()
